\node[person] at (\BobX,5.2)
    {\underline{Bob}};

% ==================================================
% Public-key generation
% ==================================================

\node[
    rotate=90,
    textline,
    font=\normalsize,
    align=center
] at (-3.0,3.65)
    {publieke sleutel\\generatie};

\node[textline] at (\AliceX,4.55)
    {kiest willekeurige};

    {$a\in\{0,1,\ldots,p-1\}$};

    {berekent publieke\\sleutel $h=g^a$};

\draw[message]
    (2.05,3.05)

    {versleutelt $m$ naar};

\draw[message]
    (7.35,-0.05)
    --
    node[above=5pt, font=\large] {$c$}
