        }
    },
    matrixB/.style={
        matrix of math nodes,
        nodes in empty cells,
        inner sep=0pt,
        row sep=-\pgflinewidth,
        column sep=-\pgflinewidth,
        nodes={
            draw=black!55,
            line width=0.55pt,
            fill=red!8,
            minimum width=1.25cm,
            minimum height=1cm,
            inner sep=0pt,
            anchor=center,
            font=\large
        }
    },
    selectedA/.style={

    selectedB/.style={

    },

% ==================================================
% --- Left matrix A ---
% ==================================================

\matrix[matrixA] (A) at (0,0) {
    a_{0,0} & a_{0,1} & |[selectedA]| a_{0,2} & a_{0,3} \\
    a_{1,0} & a_{1,1} & |[selectedA]| a_{1,2} & a_{1,3} \\
    a_{2,0} & a_{2,1} & |[selectedA]| a_{2,2} & a_{2,3} \\
    a_{3,0} & a_{3,1} & |[selectedA]| a_{3,2} & a_{3,3} \\
};

% Colored outer frame
\draw[frameA]
    (A.north west) rectangle (A.south east);

% Border around the selected third column
\node[
    columnHighlight,
    fit=(A-1-3)(A-4-3)
] (Acolumn) {};

% Label for the selected column vector
\node[
    element,
    text=blue!70!black,
    anchor=north
] at ($(Acolumn.south)+(0,-0.3)$)
    {$\boldsymbol{s}\in\left(\mathbb{F}_{2^8}\right)^4$};

% ==================================================
% --- Right matrix B ---
% ==================================================

\matrix[matrixB] (B) at (9.5,0) {
    b_{0,0} & b_{0,1} & |[selectedB]| b_{0,2} & b_{0,3} \\
    b_{1,0} & b_{1,1} & |[selectedB]| b_{1,2} & b_{1,3} \\
    b_{2,0} & b_{2,1} & |[selectedB]| b_{2,2} & b_{2,3} \\
    b_{3,0} & b_{3,1} & |[selectedB]| b_{3,2} & b_{3,3} \\
};

% Colored outer frame
\draw[frameB]
    (B.north west) rectangle (B.south east);

% Border around the selected third column
\node[
    columnHighlight,
    fit=(B-1-3)(B-4-3)
] (Bcolumn) {};

% ==================================================
% --- Transformation matrix ---
% ==================================================

\node[mixingbox] (M) at (4.75,4.3) {
    $\begin{bmatrix}
        \mathtt{0x02} & \mathtt{0x03} & \mathtt{0x01} & \mathtt{0x01} \\
        \mathtt{0x01} & \mathtt{0x02} & \mathtt{0x03} & \mathtt{0x01} \\
        \mathtt{0x01} & \mathtt{0x01} & \mathtt{0x02} & \mathtt{0x03} \\
        \mathtt{0x03} & \mathtt{0x01} & \mathtt{0x01} & \mathtt{0x02}
    \end{bmatrix}$
};

% Multiplication symbol
\node[element, text=purple!80!black]
    at ($(M.east)+(0.45,0)$)
    {$\times$};

% ==================================================
% --- Arrows ---
% ==================================================

% Selected column of A to the transformation matrix
\draw[arrow, rounded corners=3pt]
    (Acolumn.north)
    --
    ++(0,2.15)
    --
    ($(M.west)+(0,0.35)$);

% Transformation matrix to selected column of B
\draw[arrow]
    ($(M.east)+(0,-0.25)$)
    to[
        out=0,
        in=90,
        looseness=1.15
    ]
    (Bcolumn.north);
